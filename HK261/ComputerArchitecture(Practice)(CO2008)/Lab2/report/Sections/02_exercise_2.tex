\section{Exercise 2}

Các chuỗi trong \texttt{.data} đảm nhiệm lời nhắc, thông báo lỗi và hai chuỗi kết quả:

\begin{codeblock}[language=MIPS]
prompt_a: .asciiz "a = "
prompt_b: .asciiz "b = "
err_msg: .asciiz "Input must be a positive integer. Please try again.\n"
gcd_output: .asciiz "GCD = "
lcm_output: .asciiz "\nLCM = "
\end{codeblock}

Mỗi số đọc vào được kiểm tra bằng \texttt{bgtz}: lớn hơn 0 thì sang nhãn \texttt{valid\_a} lưu vào thanh ghi, ngược lại in \texttt{err\_msg} rồi quay lại chính nhãn đọc để hỏi lại. Hai khối đọc giống nhau hoàn toàn, chỉ khác chuỗi lời nhắc và thanh ghi đích:

\begin{codeblock}[language=MIPS]
read_a:
  li $v0, 4         # syscall 4: in chuỗi
  la $a0, prompt_a  # in "a = "
  syscall

  li $v0, 5         # syscall 5: đọc số nguyên
  syscall
  bgtz $v0, valid_a # a > 0 thì hợp lệ

  li $v0, 4         # syscall 4: in chuỗi
  la $a0, err_msg   # thông báo nhập sai
  syscall

  j read_a          # hỏi lại từ đầu

valid_a:
  move $s0, $v0     # $s0 = a
\end{codeblock}

GCD tính bằng vòng lặp Euclid: mỗi bước thay cặp $(a, b)$ bằng $(b, a \bmod b)$, dừng khi $b = 0$ thì $a$ là GCD. Hai thanh ghi \texttt{\$t0} và \texttt{\$t1} đóng vai trò $a$ và $b$, phần dư lấy qua \texttt{div} + \texttt{mfhi}:

\begin{codeblock}[language=MIPS]
gcd:
  move $t0, $s0          # $t0 = a
  move $t1, $s1          # $t1 = b

gcd_loop:
  beqz $t1, print_output # b == 0: $t0 là GCD

  div $t0, $t1           # chia a cho b
  mfhi $t2               # $t2 = a % b

  move $t0, $t1          # a = b
  move $t1, $t2          # b = a % b

  j gcd_loop             # lặp lại với cặp mới
\end{codeblock}

LCM tính từ GCD theo $\text{lcm}(a, b) = a \times b / \gcd(a, b)$, ngay tại nhãn \texttt{print\_output} với \texttt{mul} rồi \texttt{div} + \texttt{mflo}:

\begin{codeblock}[language=MIPS]
print_output:
  mul $t1, $s0, $s1 # $t1 = a * b
  div $t1, $t0      # chia cho GCD
  mflo $t1          # $t1 = LCM
\end{codeblock}

Kết quả được in ra:

\begin{codeblock}[language=MIPS]
  li $v0, 4          # syscall 4: in chuỗi
  la $a0, gcd_output # in "GCD = "
  syscall

  li $v0, 1          # syscall 1: in số nguyên
  move $a0, $t0      # in GCD
  syscall

  li $v0, 4          # syscall 4: in chuỗi
  la $a0, lcm_output # in "\nLCM = "
  syscall

  li $v0, 1          # syscall 1: in số nguyên
  move $a0, $t1      # in LCM
  syscall
\end{codeblock}

Nhập $a = 6$, $b = 45$:

\begin{codeblock}[language={}]
a = 6
b = 45
GCD = 3
LCM = 90
\end{codeblock}

Nhập $a = -5$ (bị từ chối), sau đó $a = 6$, $b = 45$:

\begin{codeblock}[language={}]
a = -5
Input must be a positive integer. Please try again.
a = 6
b = 45
GCD = 3
LCM = 90
\end{codeblock}
