\subsection{Bảng biểu chuẩn phong cách Học thuật (Booktabs)}

\subsection{Sơ đồ vector trực quan bằng TikZ}
Template đã nạp sẵn gói \texttt{tikz} cùng các thư viện cần thiết để vẽ sơ đồ quy trình sắc nét ngay trong file \LaTeX:

\begin{figure}[h!]
\centering
\begin{tikzpicture}[
    node distance=1.8cm,
    block/.style={rectangle, draw=blue!70, fill=blue!5, thick, rounded corners, minimum height=1cm, minimum width=3.2cm, align=center},
    arrow/.style={-Latex, thick, gray!80!black}
]
    \node [block] (input) {Dữ liệu ca thi \&\\Hồ sơ cán bộ};
    \node [block, right=of input] (solver) {Bộ giải tối ưu\\(ILP Solver)};
    \node [block, right=of solver] (output) {Lịch phân công\\hợp lệ (\texttt{.csv})};

    \draw [arrow] (input) -- node[above, font=\footnotesize] {Mô hình hóa} (solver);
    \draw [arrow] (solver) -- node[above, font=\footnotesize] {Tối ưu hóa} (output);
\end{tikzpicture}
\caption{Quy trình xử lý dữ liệu và xuất kết quả phân công}
\label{fig:pipeline}
\end{figure}
